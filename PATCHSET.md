function replaceOnce(source, before, after, label) {
  const first = source.indexOf(before);
  if (first < 0) throw new Error(`Patch context not found: ${label}`);
  if (source.indexOf(before, first + before.length) >= 0) {
    throw new Error(`Patch context is not unique: ${label}`);
  }
  return source.slice(0, first) + after + source.slice(first + before.length);
}

      failures.push(`${path} is missing`);

    const actual = sha256(read(path));
    if (actual !== expected) failures.push(`${path}: expected ${expected}, found ${actual}`);
  }

    throw new Error(
      `This patch targets main commit ${TARGET_COMMIT} (v5.1.0). Refusing to modify a different tree:\n- ${failures.join("\n- ")}`

    let source = read(path);
    for (const [before, after, label] of operations) {
      source = replaceOnce(source, before, after, label);
    }
    files[path] = source;
  }

  for (const [path, addition] of Object.entries(APPENDS)) {
    const source = read(path).replace(/\s*$/, "");
    files[path] = `${source}${addition.trimEnd()}\n`;
  }

  const pkg = JSON.parse(read("package.json"));
  pkg.version = "5.2.0";
  if (!pkg.scripts.test.includes("v5.2-regression-tests.mjs")) {
    pkg.scripts.test += " && node v5.2-regression-tests.mjs";
  }
  files["package.json"] = JSON.stringify(pkg, null, 2) + "\n";
  files["v5.2-regression-tests.mjs"] = REGRESSION_TESTS;
  return files;
}

function writeBackups(files) {
  rmSync(BACKUP, { recursive: true, force: true });
  mkdirSync(BACKUP, { recursive: true });
  for (const path of Object.keys(files)) {
    const source = resolve(ROOT, path);
    if (!existsSync(source)) continue;
    const target = resolve(BACKUP, path);
    mkdirSync(dirname(target), { recursive: true });
    copyFileSync(source, target);
  }
}

function restore(files) {
  for (const path of Object.keys(files)) {
    const backup = resolve(BACKUP, path);
    const target = resolve(ROOT, path);
    if (existsSync(backup)) copyFileSync(backup, target);
    else rmSync(target, { force: true });
  }
}

function updateChecksums() {
  const manifestPath = resolve(ROOT, "SHA256SUMS");
  if (!existsSync(manifestPath)) return;
  const manifest = readFileSync(manifestPath, "utf8").trim();
  const names = manifest
    ? manifest.split(/\r?\n/).map(line => line.trim().split(/\s+/).at(-1)).filter(Boolean)
    : [];
  if (!names.includes("v5.2-regression-tests.mjs")) names.push("v5.2-regression-tests.mjs");
  const lines = [...new Set(names)]
    .filter(name => existsSync(resolve(ROOT, name)))
    .sort()
    .map(name => `${sha256(read(name))}  ${name}`);
  writeFileSync(manifestPath, lines.join("\n") + "\n");
}
